r"""gate_common.py — phan dung chung cho MOI cong QC buoc 1 (chot 2026-09-18, P0).

Truoc day moi qc_*.py tu in "✗ ..." mot kieu, tu quyet retry mot kieu, khong luu lich su lan chay,
khong co bao cao escalation — nguoi doc phai doan lai tu log terminal. Module nay chuan hoa:

    phan bo anh/doan xen ke deu (dau hieu img_doan.txt sinh bang cong thuc, khong doc noi dung)
"""
import datetime as _dt
import json
import re
import sys
from pathlib import Path

try:
    sys.stdout.reconfigure(encoding="utf-8")
except Exception:
    pass

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
import llm_client  # noqa: E402

STATUS = ("PENDING", "RUNNING", "PASS", "WARN", "FAIL", "ESCALATED", "SKIPPED")


class QcError(RuntimeError):
    """Loi HA TANG (mang/model/JSON hong) — tach hẳn khoi phat hien noi dung."""


def ket_qua(gate, status, attempt=1, blocking=None, warnings=None, metrics=None, infra=False):
    if status not in STATUS:
        raise ValueError(f"status la {status!r}, phai thuoc {STATUS}")
    return {
        "gate": gate, "status": status, "attempt": int(attempt),
        "blocking_issues": list(blocking or []), "warnings": list(warnings or []),
        "metrics": dict(metrics or {}), "infra": bool(infra),
        "timestamp": _dt.datetime.now().isoformat(timespec="seconds"),
    }


def ghi_log(slug_dir, gate, kq):
    logs = Path(slug_dir) / "logs"
    logs.mkdir(exist_ok=True)
    n = 1 + len(list(logs.glob(f"{gate}_attempt_*.json")))
    kq = dict(kq, attempt=n)
    p = logs / f"{gate}_attempt_{n:03d}.json"
    p.write_text(json.dumps(kq, ensure_ascii=False, indent=1), encoding="utf-8")
    (logs / f"{gate}_latest.json").write_text(json.dumps(kq, ensure_ascii=False, indent=1),
                                              encoding="utf-8")
    return p


def _boc_json(txt):
    t = re.sub(r"^```(json)?|```$", "", (txt or "").strip(), flags=re.MULTILINE).strip()
    m = re.search(r"[\[{].*[\]}]", t, re.DOTALL)
    if not m:
        raise QcError(f"LLM tra ve khong phai JSON: {t[:200]}")
    try:
        return json.loads(m.group(0))
    except json.JSONDecodeError as e:
        raise QcError(f"JSON hong: {e}") from None


LLM_STATS = {"calls": 0, "tokens_in": 0, "tokens_out": 0, "latency_s": 0.0, "errors": 0}


def llm_stats_snapshot():
    return dict(LLM_STATS)


def goi_llm_json(prompt, task="analysis", max_tokens=8000, timeout=600):
    """Goi LLM re tra JSON. Cong don LLM_STATS de sau 2 tap co so that (calls/token/latency/loi)."""
    LLM_STATS["calls"] += 1
    try:
        r = llm_client.chat(prompt, task=task, max_tokens=max_tokens, temperature=0,
                            timeout=timeout, tries_per_model=2, verbose=False)
    except llm_client.LLMError as e:
        LLM_STATS["errors"] += 1
        raise QcError(f"LLM het chuoi fallback: {str(e)[:200]}") from None
    except Exception as e:
        LLM_STATS["errors"] += 1
        raise QcError(f"LLM loi {type(e).__name__}: {str(e)[:200]}") from None
    u = getattr(r, "usage", None) or {}
    LLM_STATS["tokens_in"] += int(u.get("prompt_tokens") or 0)
    LLM_STATS["tokens_out"] += int(u.get("completion_tokens") or 0)
    LLM_STATS["latency_s"] += float(getattr(r, "elapsed", 0) or 0)
    try:
        return _boc_json(r.text)
    except QcError:
        LLM_STATS["errors"] += 1
        raise


# ---------------------------------------------------------------- fingerprint (artifact stale)
META_NAME = "artifact_meta.json"
ARTIFACTS_MAC_DINH = ("img_prompts.txt", "img_doan.txt")


def sha256_file(path):
    import hashlib
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def ghi_fingerprint(slug_dir, artifacts=ARTIFACTS_MAC_DINH):
    """Stamp sau khi B5/B5.5 xong: artifact nao sinh tu script co hash nao. Chi ghi file dang ton tai."""
    d = Path(slug_dir)
    p = d / META_NAME
    meta = json.loads(p.read_text(encoding="utf-8")) if p.exists() else {"artifacts": {}}
    sha_s = sha256_file(d / "script.txt")
    meta["script_sha256"] = sha_s
    meta["stamped_at"] = _dt.datetime.now().isoformat(timespec="seconds")
    meta.setdefault("artifacts", {})
    for a in artifacts:
        f = d / a
        if f.exists():
            meta["artifacts"][a] = {"generated_from_script_sha256": sha_s, "sha256": sha256_file(f)}
    p.write_text(json.dumps(meta, ensure_ascii=False, indent=1), encoding="utf-8")
    return p


def kiem_stale(slug_dir, artifacts=ARTIFACTS_MAC_DINH):
    """Hash script hien tai != hash luc sinh artifact => STALE (blocking). mtime chi la phu tro
    khi chua co artifact_meta.json (tap cu)."""
    d = Path(slug_dir)
    s = d / "script.txt"
    if not s.exists():
        return [], []
    p = d / META_NAME
    if not p.exists():
        loi = kiem_mtime_script_vs_prompts(d)
        return loi, [] if loi else [f"chua co {META_NAME} (fingerprint) — chay `py -3.13 gate_common.py "
                                    f"--slug <tap> --stamp` sau khi xong B5/B5.5"]
    meta = json.loads(p.read_text(encoding="utf-8"))
    sha_now = sha256_file(s)
    blocking, warnings = [], []
    for a in artifacts:
        f = d / a
        if not f.exists():
            continue
        m = (meta.get("artifacts") or {}).get(a)
        if not m:
            warnings.append(f"{a} chua duoc stamp trong {META_NAME}")

        if m.get("generated_from_script_sha256") != sha_now:
            blocking.append(f"STALE: {a} sinh tu script.txt hash {str(m.get('generated_from_script_sha256'))[:10]}… "
                            f"nhung script.txt hien tai la {sha_now[:10]}… — script da sua sau khi viet {a}, "

        elif m.get("sha256") != sha256_file(f):
            warnings.append(f"{a} da sua tay sau khi stamp (script khong doi) — chay --stamp lai de ghi nhan")
    return blocking, warnings

def phan_bo_xen_ke(counts, toi_thieu=20, nguong=0.9):
    """True = so anh/doan la mau CHIA DEU co hoc: chi 1-2 gia tri khac nhau, va (hang so) hoac
    (xen ke k,k+1 deu dan >= nguong cac cap ke nhau). Duoi `toi_thieu` doan thi khong ket luan."""
    c = [int(x) for x in counts]
    if len(c) < toi_thieu:
        return False
    vals = sorted(set(c))
    if len(vals) == 1:
        return True
    if len(vals) != 2 or vals[1] - vals[0] != 1:
        return False
    xen = sum(1 for a, b in zip(c, c[1:]) if a != b)
    return xen / (len(c) - 1) >= nguong

    · phan bo lech bat thuong (nhieu doan 0 anh + co doan >6 anh).
  SEMANTIC (LLM re, WARN-only 2 tap dau; --mode fail de chan): moi loi goi = 1 chunk 8 doan lien
    tiep + cac anh da map vao do -> cham 1-5 tung anh; <=2 = lech. KHONG ma tran 130x50 (~7 loi goi/tap).
    Luon ghi metrics.semantic.would_block de sau 2 tap biet neu bat blocking thi chan bao nhieu ca.

Parser TU DOC img_doan.txt (KHONG dung sinh_neo.doc_img_doan — no tra dict, mat thu tu, nuot trung).

  py -3.13 qc_img_doan.py --slug <tap> [--mode warn|fail] [--no-llm] [--chunk 8]
exit 0 = PASS/WARN · exit 1 = FAIL (blocking hoac LLM loi ha tang — khong duoc coi la dat)
"""
import argparse
import collections
import json
import sys
from pathlib import Path

try:
    sys.stdout.reconfigure(encoding="utf-8")
except Exception:
    pass

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
import gate_common as G   # noqa: E402
import sinh_neo           # noqa: E402

GATE = "B55_mapping"


# ------------------------------------------------------------------ parser (giu thu tu + trung)
def doc_pairs(path):
    rows, malformed = [], []
    for n, ln in enumerate(Path(path).read_text(encoding="utf-8").splitlines(), 1):
        s = ln.strip()
        if not s or s.startswith("#"):
            continue
        if "|" not in s:
            malformed.append(f"dong {n}: {s[:40]!r} — malformed (thieu '|')"); continue
        ten, so = (x.strip() for x in s.split("|", 1))
        if not ten:
            malformed.append(f"dong {n}: {s[:40]!r} — malformed (ten rong)"); continue
        if not so.lstrip("-").isdigit():
            malformed.append(f"dong {n}: {s[:40]!r} — malformed (so doan khong phai so nguyen)"); continue
        rows.append({"line_no": n, "ten": ten, "so": int(so)})
    return rows, malformed


# ------------------------------------------------------------------ tang A: co hoc + suspicious
def kiem_co_hoc(rows, n_doan, n_prompt):
    blocking, suspicious = [], []
    if n_doan <= 0 or n_prompt <= 0:
        blocking.append(f"thieu required data: script co {n_doan} doan, img_prompts co {n_prompt} prompt")
        return blocking, suspicious, {"n_anh": len(rows), "n_doan": n_doan, "n_prompt": n_prompt,
                                      "phan_bo": [], "doan_0_anh": [], "chia_deu": False}
    ten = [r["ten"] for r in rows]
    if len(rows) != n_prompt:
        blocking.append(f"img_doan.txt co {len(rows)} dong nhung img_prompts.txt co {n_prompt} prompt")
    trung = [t for t, c in collections.Counter(ten).items() if c > 1]
    if trung:
        blocking.append(f"trung ten anh: {trung[:5]}")
    hop_le = {f"{i:03d}.png" for i in range(1, n_prompt + 1)}
    thieu = sorted(hop_le - set(ten))
    ngoai_id = sorted(set(ten) - hop_le)
    if thieu:
        blocking.append(f"thieu ID bat buoc ({len(thieu)}): {thieu[:6]}")
    if ngoai_id:
        blocking.append(f"ID ngoai tap hop le 001..{n_prompt:03d}.png ({len(ngoai_id)}): {ngoai_id[:6]}")
    ngoai = [(r["ten"], r["so"]) for r in rows if r["so"] < 1 or r["so"] > n_doan]
    if ngoai:
        blocking.append(f"{len(ngoai)} anh gan so doan ngoai pham vi 1..{n_doan}: {ngoai[:5]}")
    giam = [(a, b) for a, b in zip(rows, rows[1:]) if b["so"] < a["so"]]
    if giam:
        blocking.append(f"{len(giam)} cho so doan GIAM theo thu tu (anh di lui): "
                        + ", ".join(f"{a['ten']}={a['so']}->{b['ten']}={b['so']}" for a, b in giam[:4]))
    counts = [0] * n_doan
    for r in rows:
        if 1 <= r["so"] <= n_doan:
            counts[r["so"] - 1] += 1
    chia_deu = G.phan_bo_xen_ke(counts)
    if chia_deu:
        suspicious.append("SUSPICIOUS: phan bo anh/doan xen ke deu (2,3,2,3…) — dau hieu img_doan.txt sinh "
                          "bang cong thuc, khong doc noi dung; BAT BUOC chay semantic QC (khong --no-llm)")
    zero = [i + 1 for i, c in enumerate(counts) if c == 0]
    if n_doan >= 10 and len(zero) / n_doan >= 0.30 and max(counts) > 6:
        suspicious.append(f"SUSPICIOUS: phan bo lech bat thuong — {len(zero)}/{n_doan} doan 0 anh, "
                          f"co doan toi {max(counts)} anh; kiem lai bang semantic")
    metrics = {"n_anh": len(rows), "n_doan": n_doan, "n_prompt": n_prompt, "phan_bo": counts,
               "doan_0_anh": zero, "chia_deu": chia_deu}
    return blocking, suspicious, metrics

def _prompt_chunk(doan_idx, doan, items):
    ds = "\n".join(f"[doan {i}] {doan[i - 1][:400]}" for i in doan_idx)
    ps = "\n".join(json.dumps({"id": i, "file": t, "doan": d, "prompt": p[:300]}, ensure_ascii=False)
                   for i, t, d, p in items)
    return (
        "Ban la bien tap vien kiem tra video ke chuyen tieng Nhat. Duoi day la cac DOAN kich ban (co so) "
        "va cac ANH da duoc gan vao tung doan. Voi MOI anh, cham 1-5: anh co minh hoa DUNG doan da gan "
        "khong (5 = dung canh/thoi diem/nhan vat; 3 = minh hoa chung chung hop khong khi; 1 = thuoc canh "
        "KHAC HAN cua cau chuyen). Neu <=2, chi ra doan nao TRONG DANH SACH duoi day hop hon "
        "(better_doan), khong co thi null.\n"
        "Chi tra ve JSON array, moi phan tu: {\"id\": <int>, \"score\": <1-5>, \"reason\": \"<=15 tu\", "
        "\"better_doan\": <int|null>}.\n\nDOAN:\n" + ds + "\n\nANH:\n" + ps)


def kiem_semantic(rows, doan, prompts, chunk=8):
    warnings, thap, n_checked = [], [], 0
    idx_of = {r["ten"]: i + 1 for i, r in enumerate(rows)}
    for start in range(1, len(doan) + 1, chunk):
        doan_idx = list(range(start, min(start + chunk, len(doan) + 1)))
        items = [(idx_of[r["ten"]], r["ten"], r["so"], prompts[idx_of[r["ten"]] - 1])
                 for r in rows if r["so"] in doan_idx and idx_of[r["ten"]] - 1 < len(prompts)]
        if not items:
            continue
        res = G.goi_llm_json(_prompt_chunk(doan_idx, doan, items))
        if not isinstance(res, list):
            raise G.QcError("LLM khong tra array")
        for r in res:
            try:
                i, sc = int(r["id"]), int(r["score"])
            except Exception:
                continue
            if not (1 <= i <= len(rows)):
                continue
            n_checked += 1
            if sc <= 2:
                t, d = rows[i - 1]["ten"], rows[i - 1]["so"]
                bd = r.get("better_doan")
                goi_y = f" — goi y doan {bd}" if bd else ""
                warnings.append(f"{t} (doan {d}) {sc}/5: {str(r.get('reason', ''))[:80]}{goi_y}")
                thap.append({"id": i, "file": t, "doan": d, "score": sc, "better_doan": bd,
                             "reason": r.get("reason", "")})
    return warnings, thap, n_checked

        for k, v in (r.get("characters") or {}).items():
            facts["characters"].setdefault(k, {}).update(v if isinstance(v, dict) else {"note": v})
        got = {}
        for t in (r.get("timeline") or []):
            try:
                got[int(t.get("doan", 0))] = t
            except Exception:
                continue
        for i in idx:
            t = got.get(i) or {"doan": i, "facts": []}
            t["doan"] = i
            facts["timeline"].append(t)
    return facts

def kiem_prompts(prompts, rows, facts, characters, batch=15):
    warnings, conflicts, ambiguous, n_checked = [], [], [], 0
    tl = {int(t["doan"]): t for t in facts.get("timeline", [])}
    n_doan = max(tl) if tl else 1
    if rows:
        doan_of = {i + 1: r["so"] for i, r in enumerate(rows)}
    else:  # khong co img_doan.txt: uoc dai doan theo vi tri
        doan_of = {i: max(1, round(i * n_doan / max(len(prompts), 1))) for i in range(1, len(prompts) + 1)}
    ten_of = {i + 1: (rows[i]["ten"] if rows and i < len(rows) else f"{i + 1:03d}.png") for i in range(len(prompts))}
    for start in range(1, len(prompts) + 1, batch):
        ids = list(range(start, min(start + batch, len(prompts) + 1)))
        items = [(i, ten_of[i], doan_of.get(i, 0), prompts[i - 1]) for i in ids]
        ds = [doan_of.get(i, 0) for i in ids if doan_of.get(i, 0)]
        lo, hi = (max(1, min(ds) - 1), max(ds) + 1) if ds else (1, 1)
        fslice = [tl[k] for k in range(lo, hi + 1) if k in tl]
        res = G.goi_llm_json(_prompt_kiem(items, fslice, facts.get("characters", {})))
        if not isinstance(res, list):
            raise G.QcError("LLM khong tra array")
        for r in res:
            try:
                i = int(r["id"])
            except Exception:
                continue
            if not (1 <= i <= len(prompts)):
                continue
            n_checked += 1
            st = str(r.get("status", "PASS")).upper()
            c = [x for x in (r.get("conflicts") or []) if isinstance(x, dict)]
            chi_tiet = "; ".join(f"{x.get('type','?')}: {x.get('detail','')}" for x in c)[:160]
            rec = {"id": i, "file": ten_of[i], "doan": doan_of.get(i), "conflicts": c}
            if st == "FAIL":
                rec["category"] = "PROMPT_CONFLICT"
                warnings.append(f"{ten_of[i]} (doan {doan_of.get(i, '?')}) PROMPT_CONFLICT: {chi_tiet}")
                conflicts.append(rec)
            elif st == "AMBIGUOUS":
                rec["category"] = "AMBIGUOUS"
                warnings.append(f"{ten_of[i]} (doan {doan_of.get(i, '?')}) AMBIGUOUS: {chi_tiet}")
                ambiguous.append(rec)
    return warnings, conflicts, ambiguous, n_checked

# -*- coding: utf-8 -*-
r"""qc_transcript.py — cong B2Q: transcript doi thu co DAY DU khong (P0, 2026-09-18).

fetch_script.py co 3 duong du phong (api -> yt-dlp subs -> whisper) nhung khong duong nao kiem
"lay du chua": transcript cut nua chung thi B3 viet tren nua cau chuyen, B3Q --competitor van PASS.
Do tat dinh tu sidecar <competitor>.segments.json (fetch_script ghi kem tu P0):
  coverage = moc cuoi / thoi luong video   (<0.80 FAIL · <0.92 WARN · khong co thoi luong -> WARN)
  max_gap  = khoang trong dai nhat (bo 60s dau/cuoi)  (>45s WARN)
  duplicate_ratio = ti le cau trung nguyen van  (>0.15 FAIL — subtitle bi lap block)
  chars_per_min   (<120 WARN — qua thua, kha nang mat doan)
Khong co sidecar (tap cu) -> SKIPPED, exit 0.

  py -3.13 qc_transcript.py --slug <tap> [--name competitor]
"""
import argparse
import collections
import json
import sys
from pathlib import Path

try:
    sys.stdout.reconfigure(encoding="utf-8")
except Exception:
    pass

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
import gate_common as G   # noqa: E402

GATE = "B2Q"


def do(seg_json):
    segs = sorted((s for s in seg_json.get("segments", []) if isinstance(s, dict)), key=lambda s: float(s.get("start", 0)))
    dur = float(seg_json.get("video_duration_sec") or 0)
    blocking, warnings = [], []
    if not segs:
        return "FAIL", ["transcript rong"], [], {"n_seg": 0}
    last = float(segs[-1]["start"]) + float(segs[-1].get("duration", 0))
    text_all = " ".join(str(s.get("text", "")) for s in segs)
    n_chars = len(text_all.replace(" ", ""))
    coverage = None
    if dur > 0:
        coverage = round(min(last / dur, 1.0), 4)
        if coverage < 0.80:
            blocking.append(f"coverage {coverage:.0%} < 80% — transcript ket thuc som ({last:.0f}s / {dur:.0f}s)")
        elif coverage < 0.92:
            warnings.append(f"coverage {coverage:.0%} < 92%")
    else:
        warnings.append("khong co thoi luong video -> khong do duoc coverage")
    max_gap, gap_at = 0.0, 0.0
    for a, b in zip(segs, segs[1:]):
        end_a = float(a["start"]) + float(a.get("duration", 0))
        g = float(b["start"]) - end_a
        if 60 <= end_a and (dur == 0 or float(b["start"]) <= dur - 60) and g > max_gap:
            max_gap, gap_at = g, end_a
    if max_gap > 45:
        warnings.append(f"gap {max_gap:.0f}s khong co transcript tai {gap_at:.0f}s")
    texts = [str(s.get("text", "")).strip() for s in segs if len(str(s.get("text", "")).strip()) >= 8]
    dup = sum(c - 1 for c in collections.Counter(texts).values() if c > 1)
    duplicate_ratio = round(dup / max(len(texts), 1), 4)
    if duplicate_ratio > 0.15:
        blocking.append(f"duplicate_ratio {duplicate_ratio:.0%} > 15% — subtitle bi lap block")
    minutes = (dur or last) / 60
    cpm = round(n_chars / minutes, 1) if minutes else 0
    if minutes and cpm < 120:
        warnings.append(f"mat do {cpm} ky tu/phut < 120 — qua thua, kha nang mat doan")
    status = "FAIL" if blocking else ("WARN" if warnings else "PASS")
    metrics = {"n_seg": len(segs), "source": seg_json.get("source"), "video_duration": dur,
               "last_timestamp": round(last, 1), "coverage": coverage, "max_gap_sec": round(max_gap, 1),
               "duplicate_ratio": duplicate_ratio, "chars_per_min": cpm}
    return status, blocking, warnings, metrics


def main(argv=None):
    ap = argparse.ArgumentParser()
    ap.add_argument("--slug", required=True)
    ap.add_argument("--name", default="competitor")
    a = ap.parse_args(argv)
    d = Path(a.slug)
    side = d / f"{a.name}.segments.json"
    if not side.exists():
        kq = G.ket_qua(GATE, "SKIPPED", warnings=[f"khong co {side.name} (tap cu, fetch truoc P0)"])
        G.ghi_log(d, GATE, kq)
        print(f"=== {GATE}: SKIPPED (khong co sidecar)")
        return 0
    status, blocking, warnings, metrics = do(json.loads(side.read_text(encoding="utf-8")))
    G.ghi_log(d, GATE, G.ket_qua(GATE, status, blocking=blocking, warnings=warnings, metrics=metrics))
    print(f"=== {GATE}: {status} · coverage={metrics.get('coverage')} gap={metrics.get('max_gap_sec')}s "
          f"dup={metrics.get('duplicate_ratio')} cpm={metrics.get('chars_per_min')}")
    for b in blocking:
        print("  ✗", b)
    for w in warnings:
        print("  ⚠", w)
    return 1 if status == "FAIL" else 0


if __name__ == "__main__":
    sys.exit(main())
